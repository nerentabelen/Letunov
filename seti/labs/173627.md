#!/bin/bash

# Переход в целевую директорию. Если ее нет, она создается.
mkdir -p ~/8482/Gerasev/Temp/
cd ~/8482/Gerasev/Temp/ || exit

# 1. Создание дерева каталогов (Вариант 1)
# Ключ -p позволяет создавать вложенные директории одной командой.
mkdir -p Lab1/Info/{Personal,University,Hobby} Lab1/A1/B1 Lab1/A2/B2 Lab1/A3/B3
cd Lab1 || exit

# 2. Модификация каталога A2
cd A2
mkdir B4 B5
rmdir B2
cd ..

# 3. Создание файлов в Personal
cd Info/Personal
echo "Иванов Иван Иванович" > Name.txt
echo "01.01.2005" > Date.txt
echo "Школа №1" > School.txt
cd ../..

# 4. Создание файлов в University
cd Info/University
echo "МГУ, Программная инженерия" > Name.txt
echo "Математика: 90, Русский: 85. Сумма: 175" > Mark.txt
cd ../..

# 5. Создание файлов в Hobby
cd Info/Hobby
echo "Чтение, программирование" > hobby.txt
cd ../..

# 6 и 7. Копирование и удаление файлов
cp Info/Hobby/hobby.txt A2/Lab_1.txt
cd A2
cp Lab_1.txt copy_Lab_1.txt
rm copy_Lab_1.txt
cd ..

# 8. Очистка экрана
clear

# 9. Вывод содержимого файлов каталога Personal
cd Info/Personal
cat Name.txt
cat Date.txt
cat School.txt

# 10. Сортировка файлов
ls -1 | sort

# 11. Объединение файлов в all.txt и вывод на экран
cat *.txt > all.txt
cat all.txt

# 12. Редактирование all.txt
echo "Год рождения: 2005" >> all.txt
cat all.txt

# 13. Копирование all.txt в A1
cp all.txt ../../A1
cd ../..

# 14. Удаление директорий с буквой А или цифрой 2
# rm -rf рекурсивно удаляет папки без запроса подтверждения
rm -rf A1 A2 A3

# 15. Изменение строки приглашения (вывод системной даты)
# В Bash за приглашение отвечает переменная PS1, системная дата вызывается ключом \d
ex
port PS1="\d "
